class SensorPreprocessor(BaseEstimator, TransformerMixin):
    """Fit data-dependent feature filtering only on the training fold."""
    def __init__(self, missing_threshold=0.40, corr_threshold=0.98):
        self.missing_threshold = missing_threshold
        self.corr_threshold = corr_threshold

    def fit(self, X, y=None):
        X = pd.DataFrame(X).copy()

        # Missingness and constant-feature rules are learned from training data only.
        missing_rate = X.isna().mean()
        self.missing_cols_ = missing_rate[missing_rate > self.missing_threshold].index.tolist()

        X1 = X.drop(columns=self.missing_cols_, errors="ignore")
        nunique = X1.nunique(dropna=True)
        self.constant_cols_ = nunique[nunique <= 1].index.tolist()

        X2 = X1.drop(columns=self.constant_cols_, errors="ignore")

        # Median imputation statistics are learned from training data only.
        self.medians_ = X2.median(numeric_only=True)

        X3 = X2.fillna(self.medians_)

        # Redundant feature removal is learned from training data only.
        corr = X3.corr().abs()
        upper = corr.where(np.triu(np.ones(corr.shape), k=1).astype(bool))
        self.corr_cols_ = [
            column for column in upper.columns
            if any(upper[column] > self.corr_threshold)
        ]

        self.feature_names_in_ = list(X.columns)
        self.feature_names_out_ = [
            c for c in X3.columns if c not in self.corr_cols_
        ]
        return self

    def transform(self, X):
        X = pd.DataFrame(X).copy()
        X = X.drop(columns=self.missing_cols_, errors="ignore")
        X = X.drop(columns=self.constant_cols_, errors="ignore")
        X = X.fillna(self.medians_)
        X = X.drop(columns=self.corr_cols_, errors="ignore")
        return X[self.feature_names_out_]

def evaluate_model(name, search):
    best = search.best_estimator_
    y_pred = best.predict(X_test)
    y_score = best.predict_proba(X_test)[:, 1] if hasattr(best, "predict_proba") else best.decision_function(X_test)

    acc = accuracy_score(y_test, y_pred)
    prec = precision_score(y_test, y_pred, pos_label=1, zero_division=0)
    rec = recall_score(y_test, y_pred, pos_label=1, zero_division=0)
    f1 = f1_score(y_test, y_pred, pos_label=1, zero_division=0)
    auc = roc_auc_score((y_test == 1).astype(int), y_score)

    print(f"===== {name} =====")
    print("Best parameters:", search.best_params_)
    print(f"Best 5-fold CV Fail-F1: {search.best_score_:.4f}")
    print(f"Test Accuracy: {acc:.4f}")
    print(f"Test Precision (Fail): {prec:.4f}")
    print(f"Test Recall (Fail): {rec:.4f}")
    print(f"Test F1 (Fail): {f1:.4f}")
    print(f"Test ROC-AUC: {auc:.4f}\n")
    print(classification_report(
        y_test, y_pred,
        labels=[-1,1],
        target_names=["Pass (-1)", "Fail (1)"],
        zero_division=0
    ))

    cm = confusion_matrix(y_test, y_pred, labels=[-1,1])

<h3 id="Model-selection-rationale">Model selection rationale¶</h3><p>The final model is selected using the highest Fail-class F1-score from 5-fold stratified cross-validation on the training data. This avoids choosing a model based on repeated inspection of the held-out test set.</p> <p>F1 balances precision and recall for the minority Fail class. Test accuracy is still reported, but it is not used as the sole selection criterion because the dataset is strongly imbalanced. The final held-out test metrics therefore provide a cleaner estimate of generalisation performance.</p>

<h1 id="8.-Conclusion-&-Improvements">8. Conclusion & Improvements¶</h1>
  1. <p>Data quality: The raw semiconductor dataset contains a large number of sensor variables, including missing, constant and highly redundant signals. Logical feature filtering reduces unnecessary dimensionality.</p>
  2. <p>Class imbalance: Approximately 93% of entities are Pass and 7% are Fail. SMOTE is therefore applied only to training folds so the real test distribution remains unchanged.</p>
  3. <p>Leakage control: Data-dependent cleansing, imputation, scaling, dimensionality reduction and SMOTE are contained inside the modelling pipelines. This ensures that cross-validation does not learn preprocessing statistics from validation/test observations.</p>
  4. <p>Model comparison: Logistic Regression, Random Forest, SVM and Gaussian Naive Bayes are tuned using 5-fold stratified GridSearchCV. A PCA-enhanced Random Forest is included as a dimensionality-reduction enhancement.</p>
  5. <p>Model selection: The final model is selected using training cross-validation Fail-F1 rather than test-set performance. The held-out test set is used only for final reporting.</p>
  6. <p>Feature reduction: The PCA experiment directly investigates whether a compressed representation can preserve useful predictive information, addressing the project question of whether every measured signal is necessary.</p>
<h3 id="Future-improvements">Future improvements¶</h3>
  • Tune the decision threshold using a validation set if the business cost of missed Fails versus false alarms is known.
  • Add model explainability such as permutation importance or SHAP to support process-engineer investigation.
  • Validate the selected pipeline on data from a later production period to measure temporal robustness.
  • Monitor data drift and recalibrate/retrain the model as manufacturing conditions change.
<p>Final takeaway: The workflow provides a reproducible, leakage-aware machine-learning pipeline for semiconductor yield prediction, while explicitly accounting for class imbalance, high dimensionality and the need to detect Fail events</p>
